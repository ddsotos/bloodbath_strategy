cmake_minimum_required(VERSION 3.20)
project(bloodbath_research VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_library(bloodbath engine/core.cpp engine/base_work.cpp engine/progression.cpp engine/population.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS engine/progression_population_manifest.txt)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/engine/progression_population_manifest.txt PROGRESSION_POPULATION_TEXT)
string(REPLACE "\r\n" "\n" PROGRESSION_POPULATION_TEXT "${PROGRESSION_POPULATION_TEXT}")
string(SHA256 PROGRESSION_POPULATION_HASH "${PROGRESSION_POPULATION_TEXT}")
target_compile_definitions(bloodbath PUBLIC BLOODBATH_PROGRESSION_POPULATION_HASH="${PROGRESSION_POPULATION_HASH}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS engine/base_rules_manifest.txt)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/engine/base_rules_manifest.txt BASE_RULES_TEXT)
string(REPLACE "\r\n" "\n" BASE_RULES_TEXT "${BASE_RULES_TEXT}")
string(SHA256 BASE_RULES_HASH "${BASE_RULES_TEXT}")
target_compile_definitions(bloodbath PUBLIC BLOODBATH_BASE_RULES_HASH="${BASE_RULES_HASH}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS engine/toy_manifest.txt)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/engine/toy_manifest.txt TOY_RULES_TEXT)
string(REPLACE "\r\n" "\n" TOY_RULES_TEXT "${TOY_RULES_TEXT}")
string(SHA256 TOY_RULES_HASH "${TOY_RULES_TEXT}")
target_compile_definitions(bloodbath PUBLIC BLOODBATH_TOY_RULES_HASH="${TOY_RULES_HASH}")
target_include_directories(bloodbath PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(bloodbath PUBLIC cxx_std_20)
set_target_properties(bloodbath PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(bloodbath PUBLIC /W4 /WX /permissive-)
else()
  target_compile_options(bloodbath PUBLIC -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(replay_verify tools/replay_verify.cpp)
target_link_libraries(replay_verify PRIVATE bloodbath)
if(BUILD_TESTING)
  add_executable(progression_tests tests/progression_tests.cpp)
  target_link_libraries(progression_tests PRIVATE bloodbath)
  add_test(NAME progression_contracts COMMAND progression_tests)
  add_executable(population_tests tests/population_tests.cpp)
  target_link_libraries(population_tests PRIVATE bloodbath)
  add_test(NAME population_contracts COMMAND population_tests)
  add_executable(base_work_tests tests/base_work_tests.cpp)
  target_link_libraries(base_work_tests PRIVATE bloodbath)
  add_test(NAME base_work_contracts COMMAND base_work_tests)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE bloodbath)
  add_test(NAME core_contracts COMMAND core_tests)
  add_test(NAME replay_roundtrip COMMAND replay_verify)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME experiment_json COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_record.py $<TARGET_FILE:replay_verify>)
  endif()
endif()
